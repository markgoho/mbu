{{/* merit-badges/req-fallback-title.html Derives a fallback heading for a requirement whose DRG title lookup came up empty: its own leading sentence, rather than a bare "Requirement N" placeholder. mbu's DRG title coverage is sparse (most sibling groups resolve no title at all -- see req-children.html), so this fallback is the COMMON case, not a rare edge -- a placeholder that just repeats the path reads as empty boilerplate across dozens of search results in a row. The requirement's own words are always available and always distinct. Params (dict): text - the requirement's verbatim text (raw markdown) Returns: a plain-text string -- the leading sentence, capped at 80 characters on a word boundary with a trailing ellipsis when longer. */}} {{ $plain := .text | markdownify | plainify }} {{ $lead := $plain }} {{ $leadMatch := findRE `^[^.!?]*[.!?]` $plain 1 }} {{ with $leadMatch }} {{ $lead = index . 0 }} {{ end }} {{ $lead = strings.TrimSuffix "." $lead }} {{ $lead = strings.TrimSuffix "!" $lead }} {{ $lead = strings.TrimSuffix "?" $lead }} {{ $lead = strings.TrimSpace $lead }} {{ if gt (len $lead) 80 }} {{ $words := split $lead " " }} {{ $capped := "" }} {{ range $words }} {{ $candidate := strings.TrimSpace (printf "%s %s" $capped .) }} {{ if le (len $candidate) 80 }} {{ $capped = $candidate }} {{ end }} {{ end }} {{ $lead = printf "%s…" $capped }} {{ end }} {{ return $lead }}
