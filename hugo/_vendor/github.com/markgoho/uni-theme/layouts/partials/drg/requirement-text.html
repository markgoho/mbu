{{/* drg/requirement-text.html The text a drg/requirement shortcode shows. On the page's lead requirement (number == page req_number), a leading sentence that only echoes the page title (or reads as an option/short label) is dropped, when at least 10 characters remain. Shared by the shortcode's .html and .md variants (uni-theme#39). Params (dict): page, number, text (the trimmed .Inner). Returns: dict {text, is_lead}. */}} {{ $text := .text }} {{ $isLead := eq .number .page.Params.req_number }} {{ if and $text $isLead }} {{ $title := replaceRE `^(Req|Requirement) [0-9A-Za-z]+([–-][0-9A-Za-z]+)? — ` "" .page.Title }} {{ if $title }} {{ $leadSentence := findRE `^[^.!?]*[.!?]["'‘’“”)]*` $text 1 }} {{ if $leadSentence }} {{ $lead := index $leadSentence 0 }} {{ $leadText := strings.TrimRight ".!?" $lead | strings.TrimSpace }} {{ $leadLower := lower $leadText }} {{ $titleLower := lower (strings.TrimSpace $title) }} {{ $matchesTitle := false }} {{ if or (eq $leadLower $titleLower) (strings.HasPrefix $titleLower $leadLower) (strings.HasPrefix $leadLower $titleLower) }} {{ $matchesTitle = true }} {{ end }} {{ $words := split $leadText " " }} {{ $firstWord := lower (index $words 0) }} {{ $verbs := slice "do" "tell" "explain" "show" "describe" "list" "name" "discuss" "identify" "demonstrate" "find" "take" "make" "select" "choose" "visit" "prepare" "present" "complete" "outline" "interview" "review" "read" "go" "give" "keep" "write" "build" "create" "design" "conduct" "collect" "inspect" "develop" "plan" "render" "mount" "solder" "carve" "sew" "cook" "weld" }} {{ $isVerb := in $verbs $firstWord }} {{ $isOption := strings.HasPrefix $leadLower "option" }} {{ $isShortLabel := and (le (len $words) 4) (not $isVerb) }} {{ $rest := strings.TrimSpace (strings.TrimPrefix $lead $text) }} {{ $rest = replaceRE `^\s*[.!?]+\s*` "" $rest }} {{ if and (or $matchesTitle $isOption $isShortLabel) (ge (len $rest) 10) }} {{ $text = $rest }} {{ end }} {{ end }} {{ end }} {{ end }} {{ return (dict "text" $text "is_lead" $isLead) }}
