{{- if not .IsHome -}} {{- $guideLabel := site.Params.theme.guideLabel | default "Digital Resource Guide" -}} {{- /* Build ancestors list by walking up .Parent chain */ -}} {{- $ancestors := slice -}} {{- $current := . -}} {{- range seq 10 -}} {{- with $current.Parent -}} {{- $ancestors = $ancestors | append . -}} {{- $current = . -}} {{- end -}} {{- end -}} {{- $ancestors = $ancestors | collections.Reverse -}} {{- /* Build items: ancestors + current page */ -}} {{- $items := slice -}} {{- $pos := 1 -}} {{- range $ancestors -}} {{- $name := .Title -}} {{- if .IsHome -}}{{- $name = "Home" -}} {{- else if and (eq .Layout "guide") .IsSection -}}{{- $name = $guideLabel -}} {{- end -}} {{- $items = $items | append (dict "name" $name "url" (.Permalink | absURL) "position" $pos) -}} {{- $pos = add $pos 1 -}} {{- end -}} {{- $currentName := $.Title -}} {{- if and (eq $.Layout "guide") $.IsSection -}}{{- $currentName = $guideLabel -}}{{- end -}} {{- $items = $items | append (dict "name" $currentName "url" ($.Permalink | absURL) "position" $pos) -}} {{- end -}}
