{{/* drg/inherited-requirement.html Renders a child requirement whose own text is a bare topic label (e.g. "Cooking on a trail stove") that only makes sense composed onto its parent's operative verb ("Explain the safe use of a trail stove."). The merge/grammar logic (verb detection, article insertion, sentence trimming, capitalization) is a pure text transformation over two given strings -- CONTRACT.md's "no site-specific lookup inside the theme" rule doesn't touch it. Unlike mbu's local copy of this shortcode (kept, unmodified, at mbu/hugo/layouts/shortcodes/drg/inherited-requirement.html and still winning by Hugo layout precedence for mbu's own content), this theme version does NOT look up the parent's text itself via a `hugo.Data "merit-badges"`/`hugo.Data "ranks"` walk -- that is exactly the site-specific lookup CONTRACT.md forbids inside theme partials. Instead the calling site resolves `parent_text` before invoking this shortcode. For scouting-u, the Phase 3 scaffold script resolves a parent's text by `path` at generation time and writes it as a literal `parent_text="..."` param -- no runtime Hugo lookup at all. Params: number - requirement number, e.g. "6.a" topic - the child's own (possibly bare) text; falls back to .Inner if omitted parent_text - REQUIRED. The parent requirement's verbatim text, pre-resolved by the calling site. text_format - "markdown" (default) | "html". See drg/requirement.html's doc comment -- pre-rendered HTML source text needs `safeHTML`, not a second `markdownify` pass. */}} {{- $number := .Get "number" -}} {{- $topic := .Get "topic" | default (.Inner | strings.TrimSpace) -}} {{- $parentText := .Get "parent_text" | strings.TrimSpace -}} {{- $textFormat := .Get "text_format" | default "markdown" -}} {{- $displayText := $topic -}} {{- if and $parentText $topic -}} {{- $actionText := $parentText -}} {{- $actionText = strings.TrimSpace $actionText -}} {{- $actionText = replaceRE `:\s*$` "" $actionText -}} {{- $topicText := $topic -}} {{- $topicStartsWithOwnAction := false -}} {{- /* Keep in sync with OWN_ACTION_VERBS in scripts/verify-drg-requirements.ts -- a verb missing here but present there lets the verifier accept inherited-requirement for an already-self-sufficient topic, and this shortcode then wrongly prepends the parent's action anyway. */ -}} {{- if findRE `(?i)^(obtain|explain|describe|identify|name|list|tell|discuss|show|demonstrate|construct|prepare|research|record|compare|create|define|use|conduct|revisit|complete|visit|interview|pick|select|choose|learn|help|point|tour|share|make|write|draw|locate|find|outline|plan|set|review|collect|calculate|give|state|explore|diagram|establish|follow|check|render|discover|document|think|do|observe|sketch|develop|meet)\b` $topicText -}} {{- $topicStartsWithOwnAction = true -}} {{- end -}} {{- if findRE `(?i)^(with|after|before|using|by)\b` $topicText -}} {{- $topicStartsWithOwnAction = true -}} {{- end -}} {{- /* A topic with an internal sentence break ("Either...name. By sight... long.") is already a complete, multi-sentence instruction, not a bare label -- composing it onto the parent's verb produces "Then do Either by sight...". Mirrors childLooksLikeDeclarativeTeachingPrompt in verify-drg-requirements.ts. A single trailing period doesn't count. */ -}} {{- if findRE `\.\s+\S` $topicText -}} {{- $topicStartsWithOwnAction = true -}} {{- end -}} {{- if and (not $topicStartsWithOwnAction) (not (findRE `^(?i)(a|an|the)\b` $topicText)) (findRE `(?i)\b(of|for)\s*$` $actionText) -}} {{- if findRE `(?i)^[aeiou]` $topicText -}} {{- $topicText = printf "an %s" $topicText -}} {{- else if findRE `(?i)^(honest|hour|heir)` $topicText -}} {{- $topicText = printf "an %s" $topicText -}} {{- else if not (findRE `(?i)(sickness|injuries|allergies|burns|scalds|bites|stings|choking|fainting|hypoglycemia|seizures|drowning|heatstroke|frostbite|dehydration)$` $topicText) -}} {{- $topicText = printf "a %s" $topicText -}} {{- end -}} {{- end -}} {{- $actionText = replaceRE `(?i)\b(for each of|each of|the) following( [a-z][a-z ]+)?\s*$` "" $actionText -}} {{- $actionText = replaceRE `(?i)\bthese [a-z][a-z ]+\s*$` "" $actionText -}} {{- $actionText = replaceRE `(?i)\bdo the following\s*$` "" $actionText -}} {{- $actionText = strings.TrimSpace $actionText -}} {{- $actionText = replaceRE `\s+` " " $actionText -}} {{- /* Some parent requirements carry a narrative preamble before the actual instruction ("Importance of Exploration. Only the last sentence -- the one that actually ends in "the following" -- is the operative verb. */ -}} {{- $actionSentences := split $actionText ".
